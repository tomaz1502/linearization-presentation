\begin{frame}
  \frametitle{Nonlinear Arithmetic and Transcendental Functions}
  \begin{itemize}
    \item The NTA theory in SMT comprises nonlinear arithmetic and some transcendental functions
          \begin{itemize}
                  \item Trigonometric functions, $\exp$ and $\log$
          \end{itemize}
    \pause
    \vitem Applications in formal verification of dynamic systems, digital signal processing and motion planning for robots
          \begin{itemize}
                  \item Automation in proof assistants, if provided the necessary tooling
          \end{itemize}
    \pause
          \vitem Currently, no solver with complexity better than doubly-exponential for QF\_NRA, and QF\_NTA has been shown to be
          to be undecidable
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Incremental Linearization}
  \begin{itemize}
    \item \textit{Incremental linearization} is a method introduced by Cimatti et al.\ for deciding the satisfiability of formulas in QF\_NTA
          \begin{itemize}
                  \item Not a decision procedure
          \end{itemize}
    \pause
    \vitem Given a formula $\psi$ in QF\_NTA, a new formula $\psi'$ is obtained by abstracting each transcendental function and variable multiplication as an uninterpreted function
    \pause
    \vitem $\psi'$ is thus a formula in QF\_UFLRA. If $\psi'$ is unsatisfiable, then so is $\psi$
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Incremental Linearization}
  \begin{itemize}
    \item If $\psi'$ is satisfiable, not necessarily $\psi$ is. In this cast, the method incrementally introduces
          new lemmas blocking spurious solutions
    \pause
    \vfill
\begin{tcolorbox}[colback=green!5,colframe=green!40!black,title=Example]
  Let $\psi = x^{2} + y^{2} \le 2 \wedge (x \le -1.1 \lor x \ge 1.1) \wedge (y \le -1.1 \lor y \ge 1.1)$.
  $\psi'$ is the same, but instead of $x^{2}$ and $y^{2}$ we have $f_{*}(x, x)$ and $f_{*}(y, y)$.
  It is possible to interpret $f_*$ in such a way that $\psi'$ is satisfied, but the original formula is unsatisfiable.
\end{tcolorbox}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Incremental Linearization}
  \begin{itemize}
    \item Spurious solutions involving variable multiplication are blocked through the \textit{tangent plane} equation
          and some sign properties
    \pause

    \vitem Spurious solutions involving transcendental functions are blocked using their \textit{Taylor polynomial}
    \pause

  \begin{figure}
  \centering
  \includegraphics[scale=0.1]{images/tanPlane.png}
  \caption*{Source: Cimatti et al.}
  \end{figure}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Incremental Linearization}
  \begin{algorithmic}[1]
    \Function{IncrementalLinearization}{$\psi$}
      \State $\psi' \gets$ \Call{InitialAbstraction}{$\psi$}
      \State $\Gamma \gets \varnothing$
      \While{\Call{InTimeLimit}{\null}}
        \State $\langle sat, \mu \rangle \gets $\Call{SolveUFLRA}{$\psi' \wedge \Gamma$}
        \If{\textbf{not} $sat$}
          \Return UNSAT
        \EndIf
        \State $\langle sat', \Gamma' \rangle \gets $\Call{CheckRefine}{$\psi', \mu$}
        \If{$sat'$}
          \Return SAT
        \EndIf
        \State $\Gamma \gets \Gamma \cup \Gamma'$
      \EndWhile
    \EndFunction
  \end{algorithmic}
\end{frame}

\begin{frame}
  \frametitle{Proof Calculus}
  \begin{itemize}
    \item cvc5 implements a slightly modified version of this method
    \pause
    \vitem The solver also defines a \textit{proof calculus} composed of 21 lemmas capturing its reasoning
    \pause
    \vitem Types of lemmas
         \begin{itemize}
                 \item Rules eliminating spurious solutions
                 \item Phase shifting for $\sin$
                 \item Bounding $\pi$
         \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Other transcendental functions}
  \begin{itemize}
    \item $\sin$ and $\exp$ are the only functions directly handled
    \pause
    \vitem $\cos(x) = \sin(x + \frac{\pi}{2})$
    \pause

\begin{frame}
  \frametitle{Lemmas on multiplication - ARITH\_MULT\_SIGN}

\begin{center}
  $\infer[]{m \bowtie 0}{f_{1} \wedge \cdots \wedge f_{k} \mid m}$
\end{center}

\vspace{20pt}

Each $f_i$ has the form $x_j \bowtie 0$. $m$ is a monomial composed by powers of the $x_i$'s. The relation in
the conclusion is defined based on the parity of the exponents and on the sign of each $x_i$

\end{frame}

\begin{frame}[fragile]
  \frametitle{Lemmas for transcendental functions - Bounding $\exp$ and $\sin$}
\begin{center}
$\infer[]{t \geq c \rightarrow \exp(t) \geq \texttt{maclaurin}(\exp, d, c)}{d \text{ odd} \mid d,c,t}$
\end{center}

\vspace{15pt}

\begin{center}
  $\infer[]{t \in [l, u] \rightarrow
  \sin(t) \geq \texttt{secant}(\sin, l, u, t)}{d \equiv 3 \text{ mod $4^*$} \wedge (0 < l \wedge u \le \pi)^* \mid d,t,l,u}$
\end{center}

\vspace{15pt}

Where \texttt{secant} is the secant line to the $d$-th Taylor polynomial at the points $(l, p(l))$ and $(u, p(u))$

\end{frame}

\begin{frame}
  \frametitle{Taylor's theorem and Lagrange's remainder}
  \begin{itemize}
    \item These results follow from \textit{Taylor's theorem} with the Lagrange form of the remainder:

    \vfill
    \begin{center}
      $\exists x' \in [x_0, x], f(x) - \left( \mathlarger{\sum_{j=0}^{n}}\dfrac{f^{(j)}(x_0)}{j!}(x - x_0)^j \right) = \dfrac{f^{(n+1)}(x')}{(n+1)!}(x - x_0)^{n+1}$
    \end{center}
    \pause
          \vitem This formula already existed in mathlib,
          but with the additional hypothesis that $x_0 < x$. We extended it to the general version
  \end{itemize}
\end{frame}
